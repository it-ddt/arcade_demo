"""Модуль приложения."""

from collections.abc import Generator

import arcade
import pyglet

import config
import utils
from views import GameView, LoadView, MenuView

from .sound_manager import SoundManager


class App(arcade.Window):
    """Окно: менеджер представлений."""

    def __init__(self) -> None:
        """Конструктор класса.

        Создает и показывает представление геймплея;
        Запускает главный цикл движка.
        """
        # убирает рваные края у текстовых объектов
        pyglet.options.dpi_scaling = "real"

        # выбирает второй экран
        display = pyglet.display.get_display()
        screens = display.get_screens()
        screen = screens[0]

        super().__init__(
            fullscreen=True,
            update_rate=1 / config.FPS,
            draw_rate=1 / config.FPS,
            screen=screen,
        )

        self.sound_manager = SoundManager()

        self.textures: dict[arcade.Texture] = {}

        self.loader = self.load_assets()

        self.views = {
            "load": LoadView(),
        }
        self.switch_view("load")

        arcade.run()

    def switch_view(self, view_name: str) -> None:
        """Включает представление."""
        view = self.views.get(view_name)
        if not view:
            error_message = f"Представление {view_name} не найдено."
            raise RuntimeError(error_message)
        self.show_view(view)

    def load_assets(self) -> Generator[int]:
        """Загружает ассеты: текстуры и звуки."""
        percentage = 0

        textures_names = utils.get_filenames(config.IMG_DIR, config.IMG_EXTENSIONS)
        sounds_names = utils.get_filenames(config.SOUND_DIR, config.SOUND_EXTENSIONS)

        assets_count = (len(textures_names) + len(sounds_names))
        if assets_count:
            load_step = round(100 / assets_count)
        else:
            load_step = 100
        file_names = textures_names + sounds_names
        for file_name in file_names:
            name, extention = file_name.split(".")[0], file_name.split(".")[-1]
            if extention in config.IMG_EXTENSIONS:
                self.textures[name] = arcade.load_texture(config.IMG_DIR / file_name)
            elif extention in config.SOUND_EXTENSIONS:
                sound = arcade.load_sound(config.SOUND_DIR / file_name)
                self.sound_manager.add_sound(name, sound)
            percentage += load_step
            yield percentage

    def make_views(self) -> None:
        """Создает остальные представления."""
        views = {
            "menu": MenuView(),
            "game": GameView(),
        }
        self.views.update(views)

    def start_new_game(self) -> None:
        """Начинает новую игру."""
        self.switch_view("game")

    def continue_game(self) -> None:
        """Продолжает игру."""

    def on_setup(self) -> None:
        """Все ассеты загружены!."""
        self.make_views()
        self.switch_view("menu")
        self.sound_manager.play_music()

"""Модуль звука."""

from typing import TYPE_CHECKING

import arcade

if TYPE_CHECKING:
    from pyglet.media import Player


class SoundManager:
    """Менеджер звука."""

    def __init__(self) -> None:
        """Создает менеджер звуков."""
        self._volume = 1.0
        self._is_mute = not bool(self._volume)
        self.sounds: dict[str, arcade.Sound] = {}
        self.players: list[Player] = []

    @property
    def volume(self) -> float:
        """Уровень громкости."""
        return self._volume

    @volume.setter
    def volume(self, value: float) -> None:
        value = max(0.0, min(1.0, value))
        self._volume = value
        for player in self.players:
            player.volume = self._volume
        self._is_mute = not bool(self._volume)

    def add_sound(self, name: str, sound: arcade.Sound) -> None:
        """Добавляет звук."""
        self.sounds[name] = sound

    def play_sound(self, sound_name: str) -> None:
        """Издает звук один раз."""
        sound = self.sounds.get(sound_name)
        if not sound:
            return
        player = sound.play(volume=self._volume)
        self.players.append(player)

    def play_music(self) -> None:
        """Проигрывает музыку циклично."""
        sound = self.sounds.get("music")
        if not sound:
            return
        player = sound.play(loop=True, volume=self._volume)
        self.players.append(player)

    def update(self) -> None:
        """Удаляет только отыгравшие плееры (которые не играют)."""
        self.players = [player for player in self.players if player.playing]

"""Базовый спрайт - родительский для остальных."""

import math
from abc import ABC, abstractmethod

import arcade


class BaseSprite(arcade.Sprite, ABC):
    """Базовый спрайт."""

    @abstractmethod
    def __init__(self, x: int, y: int, texture: arcade.Texture) -> None:
        """Создает базовый спрайт с центром в x y и текстурой texture."""
        super().__init__()
        self.texture = texture
        self.center_x = x
        self.center_y = y
        self.motion_speed = 200
        self.rotation_speed = 100
        self.turn_direction = 0
        self.move_direction = 0

    def update(self, delta_time: float) -> None:
        """Обновление.

        Перемещение и поворот.
        """
        self.angle += self.turn_direction * self.rotation_speed * delta_time
        angle_rad = math.radians(self.angle)

        self.center_x += (
            math.sin(angle_rad)
            * self.move_direction
            * self.motion_speed
            * delta_time
        )
        self.center_y += (
            math.cos(angle_rad)
            * self.move_direction
            * self.motion_speed
            * delta_time
        )

    def stay_inside_bounds(self, w_width: int, w_height: int) -> None:
        """Не дает спрайту выйти за границы экрана."""
        if self.center_x > w_width:
            self.center_x = w_width
        elif self.center_x < 0:
            self.center_x = 0

        if self.center_y > w_height:
            self.center_y = w_height
        elif self.center_y < 0:
            self.center_y = 0

"""Модуль кликабельной кнопки."""

from collections.abc import Callable

import arcade


class Button(arcade.SpriteSolidColor):
    """Кнопка."""

    def __init__(
            self,
            x: int,
            y: int,
            width: int = 600,
            height: int = 100,
            color: arcade.color.Color = arcade.color.BABY_BLUE,
            text: str = "кнопка",
            action: Callable | None = None,
    ) -> None:
        """Создает кнопку с центром в x y."""
        super().__init__(width, height, x, y, color)
        self.action = action
        self.text = text
        self.label = arcade.Text(self.text, x, y, anchor_x="center", anchor_y="center", font_size=50)

    def on_click(self) -> None:
        """Вызов действия по ЛКМ."""
        self.action()

    def on_draw(self) -> None:
        """Рисует прямоугольник (спрайт) и текст."""
        arcade.draw_sprite(self)
        self.label.draw()

"""Модуль спрайта мыши."""

import math

import arcade

from .base_sprite import BaseSprite


class Mouse(BaseSprite):
    """Мышь."""

    def __init__(self, x: int, y: int, texture: arcade.Texture) -> None:
        """Создает мышь с центром в x y с текстурой texture."""
        super().__init__(x, y, texture)
        self.motion_speed = 200
        self.rotation_speed = 300
        self.flee_distance = 200

    def __str__(self) -> str:
        """Строковое представление."""
        return f"Мышь в {self.center_x} {self.center_y}"

    def move(self, player_x: int, player_y: int) -> None:
        """Бегство или стояние на месте."""
        d_x = self.center_x - player_x
        d_y = self.center_y - player_y
        distance_to_player = math.hypot(d_x, d_y)
        if distance_to_player <= self.flee_distance:  # бегство
            angle_to_player = math.degrees(math.atan2(d_x, d_y))
            target_angle = angle_to_player + 180

            # определить, нужно ли повернуться
            diff = (target_angle - self.angle) % 360 - 180
            angle_threshold = 5
            if abs(diff) < angle_threshold:
                self.turn_direction = 0
                self.move_direction = 1
            elif diff > 0:
                self.turn_direction = 1
                self.move_direction = 0
            else:
                self.turn_direction = -1
                self.move_direction = 0
        else:
            self.move_direction = 0
            self.turn_direction = 0

"""Утилиты."""

import pathlib

import config


def get_scale(
        width_required: float,
        height_required: float,
        width_initial: float,
        height_initial: float,
    ) -> float:
    """Возвращает масштаб.

    Масштаб - множитель: наибольше из отношений
    требуемой к исходной ширине (высоте).
    """
    width_ratio = width_required / width_initial
    height_ratio = height_required / height_initial
    return max(width_ratio, height_ratio)


def get_filenames(path: pathlib.Path, extentions: tuple[str]) -> list[str]:
    """Возвращает имена и расширения файлов в папке."""
    files = pathlib.Path.iterdir(path)
    return [
        file.name for file in files
        if file.is_file()
        and file.suffix.lower()[1:] in extentions
    ]

"""Модуль базового представления - родительского для всех."""

from __future__ import annotations

from abc import ABC, abstractmethod
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from main import App
    from sprites import BaseSprite, Button

import arcade

import utils


class BaseView(arcade.View, ABC):
    """Базовое представление."""

    window: App

    @abstractmethod
    def __init__(self, bg_name: str = "") -> None:
        """Создает базовое представление."""
        super().__init__()
        self.bg_sprites = arcade.SpriteList()
        self.sprites: arcade.SpriteList[BaseSprite] = arcade.SpriteList()
        self.buttons: arcade.SpriteList[Button] = arcade.SpriteList()
        self.text_objects: list[arcade.Text] = []
        if bg_name:
            self.make_bg(bg_name)

    def make_bg(self, bg_name: str) -> None:
        """Создает фон."""
        bg_x = self.window.width // 2
        bg_y = self.window.height // 2
        bg_texture = self.window.textures[bg_name]
        bg = arcade.Sprite()
        bg.texture = bg_texture
        bg.center_x = bg_x
        bg.center_y = bg_y
        bg.scale = utils.get_scale(
            self.window.width,
            self.window.height,
            bg.texture.width,
            bg.texture.height,
        )
        self.bg_sprites.append(bg)

    def on_draw(self) -> None:
        """Отрисовывает фоны, спрайты, кнопки и текстовые объекты."""
        self.clear()
        self.bg_sprites.draw()
        self.sprites.draw()
        for button in self.buttons:
            button.on_draw()
        for text_object in self.text_objects:
            text_object.draw()

    def on_update(self, _: float) -> None:
        """Обновляет менеджер звуков."""
        self.window.sound_manager.update()

"""Модуль представлений."""

import random

import arcade

from sprites import BaseSprite, Mouse, Player

from .base_view import BaseView


class GameView(BaseView):
    """Представление геймплея."""

    def __init__(self) -> None:
        """Конструктор класса.

        Создает спрайтлисты: все (для отрисовки), мыши (для столкновений);
        Создает спрайты: игрок и фон;
        Подгоняет фон к размеру окна (не оставляет пустых областей).
        """
        super().__init__("game_bg")
        self.sprites: arcade.SpriteList[BaseSprite] = arcade.SpriteList()
        self.mice: arcade.SpriteList[Mouse] = arcade.SpriteList()
        self.text_objects = list[arcade.Text]

        self.player: Player | None = None

        self.mice_spawn_interval = 1  # сек
        self.mice_spawn_timer = self.mice_spawn_interval
        self.mice_caught_counter = 0

        self.score = arcade.Text(
            str(self.mice_caught_counter),
            self.window.width // 2,
            int(self.window.height * 0.9),
            font_size=100,  # TODO: рассчитать от размера окна
            anchor_x="center",
            anchor_y="center",
        )
        self.text_objects.append(self.score)

    def on_key_press(self, symbol: int, _: int) -> None:
        """Система ввода: нажатие клавиш."""
        if symbol == arcade.key.ESCAPE:
            self.window.switch_view("menu")

        if symbol == arcade.key.D:
            self.player.turn_direction = 1
        if symbol == arcade.key.A:
            self.player.turn_direction = -1

        if symbol == arcade.key.W:
            self.player.move_direction = 1
        if symbol == arcade.key.S:
            self.player.move_direction = -1

    def on_key_release(self, symbol: int, _: int) -> None:
        """Система ввода: отпускание клавиш."""
        if symbol == arcade.key.D:
            self.player.turn_direction = 0
        if symbol == arcade.key.A:
            self.player.turn_direction = 0

        if symbol == arcade.key.W:
            self.player.move_direction = 0
        if symbol == arcade.key.S:
            self.player.move_direction = 0

    def on_update(self, delta_time: float) -> None:
            """Спавнит мышей, двигает игрока, удаляет пойманных мышей."""
            self.mice_spawn_timer += delta_time
            if self.mice_spawn_timer >= self.mice_spawn_interval:
                self.spawn_mouse()
                self.mice_spawn_timer = 0
            for mouse in self.mice:
                mouse.move(self.player.center_x, self.player.center_y)
            self.sprites.update(delta_time)
            for sprite in self.sprites:
                sprite.stay_inside_bounds(self.window.width, self.window.height)

            mice_caught: list[arcade.Sprite] = arcade.check_for_collision_with_list(
                self.player,
                self.mice,
            )
            for mouse in mice_caught:
                self.window.sound_manager.play_sound("catch")
                mouse.remove_from_sprite_lists()
                self.mice_caught_counter += 1
                self.score.text = str(self.mice_caught_counter)

    def spawn_mouse(self) -> None:
        """Спавн мыши.

        Создает спрайт с текстурой в случайных координатах
        (отступ 10% от каждого края).
        """
        texture = self.window.textures["mouse"]
        x = random.randint(
            int(self.window.width * 0.1),
            int(self.window.width * 0.9),
        )
        y = random.randint(
            int(self.window.height * 0.1),
            int(self.window.height * 0.9),
        )
        mouse = Mouse(x, y, texture)
        mouse.scale = 0.1  # TODO: считать размер от размера окна
        self.sprites.append(mouse)
        self.mice.append(mouse)

"""Модуль представления меню."""

from __future__ import annotations

from functools import partial
from typing import TYPE_CHECKING

import arcade

if TYPE_CHECKING:
    from main import App

from sprites import Button

from .base_view import BaseView


class MenuView(BaseView):
    """Меню."""

    window: App


    def __init__(self) -> None:
        """Конструктор класса.

        Создает и показывает текст.
        """
        super().__init__("menu_bg")

        self.title = arcade.Text(
            "",
            self.window.width // 2,
            self.window.height // 2,
            anchor_x="center",
            anchor_y="center",
            font_size=50,  # TODO: рассчитать от размера окна
        )
        self.text_objects.append(self.title)
        self.make_buttons()

    def on_key_press(self, symbol: int, _: int) -> None:
        """Включает представление игры по ENTER."""
        if symbol == arcade.key.ENTER and self.window.views.get("game"):
            self.window.switch_view("game")
        elif symbol == arcade.key.ESCAPE:
            arcade.exit()
        elif symbol == arcade.key.M:
            self.window.sound_manager.toggle_mute()
            self.set_menu_text()
        elif symbol == arcade.key.N:
            self.window.start_new_game()

    def on_mouse_press(self, x: int, y: int, button: int, _: int) -> None:
        """Клик на кнопки."""
        if button != arcade.MOUSE_BUTTON_LEFT:
            return
        for button_sprite in self.buttons:
            if button_sprite.collides_with_point((x, y)):
                button_sprite.on_click()
                self.make_buttons()

    def make_buttons(self) -> None:
        """Создает кнопки."""
        self.buttons.clear()
        buttons_actions = {"новая игра": self.window.start_new_game}

        if self.window.sound_manager.volume > 0.0:
            buttons_actions["выключить звук"] = partial(
                setattr, self.window.sound_manager, "volume", 0.0,
            )
        else:
            buttons_actions["включить звук"] = partial(
                setattr, self.window.sound_manager, "volume", 1.0,
            )

        if self.window.views.get("game"):  # TODO: проверить стаус игры приложения
            buttons_actions["продолжить игру"] = self.window.continue_game

        buttons_actions["выйти"] = arcade.exit

        button_height = 100  # TODO: забрать из кнопки
        offset_y = button_height // 2
        x = self.window.width // 2

        total_buttons_height = (
            button_height * len(buttons_actions)
            + (len(buttons_actions) - 1)
            * offset_y
        )

        y = self.window.height // 2 + total_buttons_height // 2 - button_height // 2

        for text, action in buttons_actions.items():
            button = Button(x, y, text=text, action=action)
            self.buttons.append(button)
            y -= button.height + offset_y
